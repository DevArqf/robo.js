## Example

```ts
// In an API route handler
export default function(request, reply) {
  // Create a JSON response
  return RoboResponse.json({
    success: true,
    message: 'Data processed successfully',
    timestamp: new Date()
  }, {
    status: 200,
    headers: {
      'X-Custom-Header': 'some-value'
    }
  })
}
```
